# ---- bin/devai-agent: gpu_flags().
def gpu_flags(runtime: str, cpu_only: bool, gpu_vendor: str = "nvidia") -> list[str]:
    if cpu_only:
        return []
    if "podman" in runtime:
        device = "amd.com/gpu=all" if gpu_vendor == "amd" else "nvidia.com/gpu=all"
        return ["--device", device, "--security-opt=label=disable"]
    return ["--gpus", "all"]
